    "label": "mug",                                      
    "description": "red ceramic mug on the table",
    "color": "red",
    "material": "ceramic",
    "x_min": 0.42,
    "x_max": 0.50,
    "y_min": 1.10,
    "y_max": 1.18,
    "z_min": 0.76,

  }'

    "max_distance": 0.8,
    "min_similarity": 0.75,
    "dry_run": false
  }'

  -H "Content-Type: application/json" \
  -d '{}'

  -H "Content-Type: application/json" \
  -d '{"uncertain_only": false, "room_id": "", "label_filter": "table"}'

Per navigare su habitat dopo aver avviato habitat_nav:   
  
ros2 topic pub /habitat/action std_msgs/msg/String "{data: 'turn_left'}" --once


ros2 topic pub /habitat/cmd_vel geometry_msgs/msg/Twist "{linear: {x: 0.3}, angular: {z: 0.5}}" --once

ros2 topic pub /habitat/cmd_vel geometry_msgs/msg/Twist "{linear: {x: 0.0}, angular: {z: 0.0}}" --once
